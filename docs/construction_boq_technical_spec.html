Project Scaffolding
Exact commands to initialise the project from zero. Run in order.

1.1 — Create Tauri + React + TypeScript Project

$ npm create tauri-app@latest boq-calculator
# When prompted, select:
# Framework: React
# Language: TypeScript
# Package manager: npm

$ cd boq-calculator
$ npm install

1.2 — Install Frontend Dependencies

# Core UI
$ npm install tailwindcss @tailwindcss/vite postcss autoprefixer
$ npm install zustand immer
$ npm install zod

# AG Grid (use Community — free)
$ npm install ag-grid-react ag-grid-community

# Export
$ npm install @react-pdf/renderer
$ npm install xlsx

# Utilities
$ npm install date-fns nanoid clsx
$ npm install @radix-ui/react-dialog @radix-ui/react-dropdown-menu @radix-ui/react-select @radix-ui/react-tooltip
$ npm install lucide-react

# Rust-side plugins (add to src-tauri/Cargo.toml)
$ cargo add tauri-plugin-sql --features sqlite
$ cargo add tauri-plugin-fs

# Frontend bindings
$ npm install @tauri-apps/plugin-sql @tauri-apps/plugin-fs @tauri-apps/plugin-dialog

// ── Project ──────────────────────────────────────────────────
export type ProjectStatus = 'draft' | 'submitted' | 'awarded' | 'archived'

export type ProjectType =
  | 'residential'
  | 'commercial'
  | 'industrial'
  | 'infrastructure'
  | 'renovation'
  | 'other'

export interface Project {
  id: number
  name: string
  clientName: string
  siteAddress: string
  projectType: ProjectType
  status: ProjectStatus
  currency: string            // 'PKR' default
  taxRate: number             // 0.17 = 17% GST
  contingencyRate: number    // 0.05 = 5% contingency, 0 = none
  profitMarginRate: number   // 0.10 = 10% profit, 0 = none
  notes: string
  createdAt: string           // ISO datetime string
  updatedAt: string
}

export type CreateProjectInput = Omit<Project, 'id' | 'createdAt' | 'updatedAt'>

// ── BOQ Item ──────────────────────────────────────────────────
export interface BoqItem {
  id: number | string        // number when saved, nanoid string when new/unsaved
  projectId: number
  revisionId: number | null  // null = current working copy
  itemNo: string              // '1', '1.1', 'A', etc.
  description: string
  unit: string                // from VALID_UNITS constant
  quantity: number
  rate: number                // PKR per unit
  amount: number              // computed: quantity × rate
  isSectionHeader: boolean
  sortOrder: number
  materialId: number | null  // FK to rates table
  notes: string
  _isDirty: boolean           // unsaved changes flag (frontend only, not in DB)
  _hasError: boolean          // validation error flag (frontend only)
  _errorMessage: string      // (frontend only)
}

// ── Rate (Material Library entry) ────────────────────────────
export interface Rate {
  id: number
  name: string
  category: RateCategory
  unit: string
  rate: number
  city: PakistanCity | null   // null = national average
  lastUpdated: string

}

// ─── src/store/boqStore.ts ────────────────────────────────────
import { create } from 'zustand'
import { immer } from 'zustand/middleware/immer'

interface BoqState {
  // ── Data ──────────────────────────────────────────────
  currentProjectId: number | null
  items: BoqItem[]              // in-memory working copy, sorted by sortOrder
  totals: BoqTotals | null     // recomputed whenever items change

  // ── UI State ──────────────────────────────────────────
  isLoading: boolean
  isSaving: boolean            // true during DB write
  lastSavedAt: Date | null
  selectedRowIds: (number | string)[]
  saveError: string | null

  // ── Actions ───────────────────────────────────────────
  loadProject: (projectId: number) => Promise<void>
  addItem: (afterId?: number | string) => void   // adds blank row, triggers autosave
  addSectionHeader: (afterId?: number | string) => void
  updateItem: (id: number | string, changes: Partial<BoqItem>) => void
  deleteItems: (ids: (number | string)[]) => Promise<void>
  duplicateItem: (id: number | string) => void
  reorderItems: (newOrder: (number | string)[]) => Promise<void>
  applyTemplate: (template: Template, mode: 'replace' | 'append') => Promise<void>
  forceSave: () => Promise<void>  // called on Ctrl+S and before export
  recalcTotals: () => void        // called after any item change
}

SQL Migrations
Complete CREATE TABLE statements. Run via migration runner in src/lib/migrations.ts on app startup.
-- migration_001_initial_schema.sql

CREATE TABLE IF NOT EXISTS projects (
  id               INTEGER PRIMARY KEY AUTOINCREMENT,
  name             TEXT NOT NULL,
  client_name      TEXT NOT NULL DEFAULT '',
  site_address     TEXT NOT NULL DEFAULT '',
  project_type     TEXT NOT NULL DEFAULT 'residential',
  status           TEXT NOT NULL DEFAULT 'draft',
  currency         TEXT NOT NULL DEFAULT 'PKR',
  tax_rate         REAL NOT NULL DEFAULT 0.17,
  contingency_rate REAL NOT NULL DEFAULT 0.0,
  profit_margin_rate REAL NOT NULL DEFAULT 0.0,
  notes            TEXT NOT NULL DEFAULT '',
  created_at       TEXT NOT NULL DEFAULT (datetime('now')),
  updated_at       TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS boq_items (
  id                INTEGER PRIMARY KEY AUTOINCREMENT,
  project_id        INTEGER NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  revision_id       INTEGER REFERENCES revisions(id) ON DELETE SET NULL,
  item_no           TEXT NOT NULL DEFAULT '',
  description       TEXT NOT NULL DEFAULT '',
  unit              TEXT NOT NULL DEFAULT 'm³',
  quantity          REAL NOT NULL DEFAULT 0,
  rate              REAL NOT NULL DEFAULT 0,
  amount            REAL NOT NULL DEFAULT 0,  -- stored for fast reads
  is_section_header INTEGER NOT NULL DEFAULT 0, -- 1=true, 0=false
  sort_order        INTEGER NOT NULL DEFAULT 0,
  material_id       INTEGER REFERENCES rates(id) ON DELETE SET NULL,
  notes             TEXT NOT NULL DEFAULT ''
);
CREATE INDEX idx_boq_project ON boq_items(project_id, sort_order);
CREATE INDEX idx_boq_revision ON boq_items(revision_id);

CREATE TABLE IF NOT EXISTS rates (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  name         TEXT NOT NULL,
  category     TEXT NOT NULL DEFAULT 'civil',
  unit         TEXT NOT NULL,
  rate         REAL NOT NULL DEFAULT 0,
  city         TEXT,                            -- NULL = national
  last_updated TEXT NOT NULL DEFAULT (datetime('now')),
  is_system    INTEGER NOT NULL DEFAULT 0
);
CREATE INDEX idx_rates_category ON rates(category, city);
CREATE INDEX idx_rates_name ON rates(name);

CREATE TABLE IF NOT EXISTS templates (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  name         TEXT NOT NULL,
  description  TEXT NOT NULL DEFAULT '',
  project_type TEXT NOT NULL DEFAULT 'residential',
  items_json   TEXT NOT NULL DEFAULT '[]',    -- JSON array of item objects
  is_system    INTEGER NOT NULL DEFAULT 0,
  created_at   TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS revisions (
  id           INTEGER PRIMARY KEY AUTOINCREMENT,
  project_id   INTEGER NOT NULL REFERENCES projects(id) ON DELETE CASCADE,
  name         TEXT NOT NULL,                 -- 'Rev A', 'Rev B', 'Client Review', etc.
  notes        TEXT NOT NULL DEFAULT '',
  grand_total  REAL NOT NULL DEFAULT 0,      -- snapshot total at time of revision
  created_at   TEXT NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS settings (
  key   TEXT PRIMARY KEY,
  value TEXT NOT NULL DEFAULT ''
);

CREATE TABLE IF NOT EXISTS migrations (
  id         INTEGER PRIMARY KEY AUTOINCREMENT,
  version    INTEGER NOT NULL UNIQUE,
  applied_at TEXT NOT NULL DEFAULT (datetime('now'))
);

7.1 — DB Singleton Pattern

// ─── src/lib/db.ts ────────────────────────────────────────────
import Database from '@tauri-apps/plugin-sql'

let _db: Database | null = null

export async function getDb(): Promise<Database> {
  if (!_db) {
    _db = await Database.load('sqlite:boq.db')
    await runMigrations(_db)
  }
  return _db
}

// Row type (snake_case from SQLite) — separate from domain type (camelCase)
interface ProjectRow {
  id: number; name: string; client_name: string
  site_address: string; project_type: string
  status: string; currency: string; tax_rate: number
  contingency_rate: number; profit_margin_rate: number
  notes: string; created_at: string; updated_at: string
}

// Mapper: DB row → domain type
export function rowToProject(row: ProjectRow): Project {
  return {
    id: row.id, name: row.name, clientName: row.client_name,
    siteAddress: row.site_address, projectType: row.project_type as ProjectType,
    status: row.status as ProjectStatus, currency: row.currency,
    taxRate: row.tax_rate, contingencyRate: row.contingency_rate,
    profitMarginRate: row.profit_margin_rate, notes: row.notes,
    createdAt: row.created_at, updatedAt: row.updated_at,
  }
}

Layout Rules

Main area
AG Grid fills flex: 1. Grid height: calc(100vh - toolbar_height - statusbar_height). Never use a fixed height. The grid must fill all available vertical space.
Summary sidebar
Fixed width: 200px. Collapsible with a toggle button. Scrolls independently from the grid. Updates in real-time on every cell edit via recalcTotals().

Title Bar Indicator

The window title must show save state: "Project Name — BOQ Editor ● Saved" (green dot) or "Project Name — BOQ Editor ○ Saving..." (dim dot). Use Tauri's window.setTitle() via invoke. This mimics VS Code's unsaved indicator — extremely important for user trust.

// ─── src/lib/calculator.ts ────────────────────────────────────

/** Round to 2 decimal places (half-up) */
export function round2(n: number): number {
  return Math.round(n * 100) / 100
}

/** Calculate amount for a single item */
export function calcItemAmount(quantity: number, rate: number): number {
  if (!isFinite(quantity) || !isFinite(rate)) return 0
  return round2(quantity * rate)
}

/** Calculate all totals for a project */
export function calcTotals(
  items: BoqItem[],
  project: Pick<Project, 'taxRate' | 'contingencyRate' | 'profitMarginRate'>
): BoqTotals {
  // Build sections array
  const sections: { label: string; subtotal: number }[] = []
  let currentSection: string = 'General'
  let currentSubtotal = 0
  let sectionStarted = false

  items.forEach((item) => {
    if (item.isSectionHeader) {
      if (sectionStarted) {
        sections.push({ label: currentSection, subtotal: round2(currentSubtotal) })
        currentSubtotal = 0
      }
      currentSection = item.description || 'Untitled Section'
      sectionStarted = true
    } else {
      currentSubtotal += item.amount
    }
  })
  if (sectionStarted) {
    sections.push({ label: currentSection, subtotal: round2(currentSubtotal) })
  }

  const subtotal = round2(sections.reduce((s, sec) => s + sec.subtotal, 0))
  const contingencyAmount = round2(subtotal * (project.contingencyRate || 0))
  const netTotal = round2(subtotal + contingencyAmount)
  const taxAmount = round2(netTotal * (project.taxRate || 0))
  const profitAmount = round2(netTotal * (project.profitMarginRate || 0))
  const grandTotal = round2(netTotal + taxAmount + profitAmount)

  return { sections, subtotal, contingencyAmount, netTotal, taxAmount, profitAmount, grandTotal }
}

// Pre-export validation — called before PDF/Excel export
export function validateBoqForExport(items: BoqItem[], project: Project): string[] {
  const errors: string[] = []
  if (!project.name) errors.push('Project name is required')
  const nonHeaderItems = items.filter(i => !i.isSectionHeader)
  if (nonHeaderItems.length === 0) errors.push('BOQ has no items to export')
  const emptyDescriptions = nonHeaderItems.filter(i => !i.description.trim())
  if (emptyDescriptions.length > 0)
    errors.push(`${emptyDescriptions.length} item(s) have no description`)
  return errors
}

export async function exportToPdf(project: Project, items: BoqItem[], settings: AppSettings) {
  // 1. Validate — throw if errors
  const errors = validateBoqForExport(items, project)
  if (errors.length) throw new Error(errors.join(', '))

  // 2. Load logo as base64 (if logoPath set)
  const logoBase64 = settings.logoPath
    ? await readFileAsBase64(settings.logoPath)
    : null

  // 3. Calc totals
  const totals = calcTotals(items, project)

  // 4. Render PDF using @react-pdf/renderer
  const blob = await pdf(
    <PdfDocument project={project} items={items} totals={totals}
      settings={settings} logoBase64={logoBase64} />
  ).toBlob()

  // 5. Show native Save dialog
  const savePath = await invoke<string | null>('save_file_dialog', {
    defaultName: `BOQ_${project.name}_${format(new Date(), 'yyyy-MM-dd')}.pdf`,
    filters: [{ name: 'PDF', extensions: ['pdf'] }]
  })
  if (!savePath) return  // user cancelled

  // 6. Write file
  const buffer = await blob.arrayBuffer()
  await invoke('write_binary_file', { path: savePath, data: Array.from(new Uint8Array(buffer)) })

  // 7. Auto-backup if enabled
  if (settings.autoBackupOnExport) await backupDatabase(settings)
}

import * as XLSX from 'xlsx'

export async function exportToExcel(project: Project, items: BoqItem[]) {
  const wb = XLSX.utils.book_new()

  // ── Row data ─────────────────────────────────────────────────
  // Row 1: Title
  // Row 2: Project name, client, site, date
  // Row 3: empty
  // Row 4: Headers — Item No | Description | Unit | Qty | Rate | Amount
  // Row 5+: items (section headers as merged full-width rows)
  // After last item: empty row, then totals

  const rows: any[][] = [
    ['BILL OF QUANTITIES', '', '', '', '', ''],
    ['Project:', project.name, '', 'Client:', project.clientName, ''],
    ['Date:', format(new Date(), 'dd/MM/yyyy'), '', 'Site:', project.siteAddress, ''],
    [],
    ['Item No', 'Description', 'Unit', 'Quantity', 'Rate (PKR)', 'Amount (PKR)'],
  ]

  let dataRowStart = 6  // 1-indexed row number of first item
  let rowIdx = dataRowStart

  items.forEach(item => {
    if (item.isSectionHeader) {
      rows.push([item.description, '', '', '', '', ''])
      // will merge A:F for this row later
    } else {
      rows.push([
        item.itemNo,
        item.description,
        item.unit,
        item.quantity,
        item.rate,
        { f: `D${rowIdx}*E${rowIdx}` },  // ← actual Excel formula!
      ])
    }
    rowIdx++
  })

  // Totals
  rows.push([])
  rows.push(['', '', '', '', 'Sub Total', { f: `SUM(F${dataRowStart}:F${rowIdx-1})` }])
  rows.push(['', '', '', '', `GST ${project.taxRate * 100}%`, { f: `F${rowIdx+1}*${project.taxRate}` }])
  rows.push(['', '', '', '', 'Grand Total', { f: `F${rowIdx+1}+F${rowIdx+2}` }])

  const ws = XLSX.utils.aoa_to_sheet(rows)

  // Column widths
  ws['!cols'] = [
    { wch: 10 }, { wch: 45 }, { wch: 8 }, { wch: 12 }, { wch: 15 }, { wch: 18 }
  ]

  XLSX.utils.book_append_sheet(wb, ws, 'BOQ')

  // Write and trigger save dialog
  const buffer = XLSX.write(wb, { bookType: 'xlsx', type: 'array' })
  // → invoke save_file_dialog + write_binary_file (same as PDF export)
}

Autosave Pattern
// In boqStore.ts updateItem action:
const saveTimer = { current: null as ReturnType<typeof setTimeout> | null }

updateItem: (id, changes) => {
  set((state) => {
    const idx = state.items.findIndex(i => i.id === id)
    if (idx === -1) return
    Object.assign(state.items[idx], changes)
    // Recompute amount immediately
    if ('quantity' in changes || 'rate' in changes) {
      state.items[idx].amount = calcItemAmount(
        state.items[idx].quantity,
        state.items[idx].rate
      )
    }
    state.items[idx]._isDirty = true
  })
  // Recalc totals synchronously
  get().recalcTotals()
  // Debounced DB save
  if (saveTimer.current) clearTimeout(saveTimer.current)
  saveTimer.current = setTimeout(async () => {
    const dirtyItems = get().items.filter(i => i._isDirty)
    await saveItemsBatch(dirtyItems)  // write all dirty items in one transaction
    set((s) => { s.lastSavedAt = new Date(); s.isSaving = false })
  }, AUTOSAVE_DELAY_MS)
}
